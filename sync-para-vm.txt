def _opt_decimal(v) -> Decimal | None:

    except (InvalidOperation, ValueError):
        raise HTTPException(422, f"valor numÃ©rico invÃ¡lido: {v!r}")

    """Devolve o PDF original do Blob, usado pelo viewer da tela de revisÃ£o.

def _field(fields: dict, name: str):
    """prebuilt-invoice field values, tolerant of SDK version differences."""
    f = fields.get(name) if fields else None
    if f is None:
        return None
    for attr in ("value_string", "value_date", "value_number", "value_integer"):
        v = getattr(f, attr, None)
        if v is not None:
            return str(v) if attr == "value_date" else v

    if currency is not None:

class DocumentIntelligenceExtractor(DocumentExtractor):
    """prebuilt-invoice pre-pass. Returns the structured fields it can find plus
    the full document text, which the structurer mines for the fields the

    def __init__(self, endpoint: str, key: str | None = None) -> None:

        else:
            from azure.identity import DefaultAzureCredential

            endpoint=endpoint, credential=credential
        )

    def extract(self, content: bytes) -> dict[str, Any]:
        poller = self._client.begin_analyze_document(
            self.MODEL_ID, body=content, content_type="application/octet-stream"
        )
        result = poller.result()

        invoices = []
        confidences = []

            fields = document.fields or {}

            items_field = fields.get("Items")

                item_fields = getattr(entry, "value_object", None) or {}

from abc import ABC, abstractmethod

from azure.core.exceptions import ResourceExistsError
from azure.storage.blob import BlobServiceClient

class BlobStorage(ABC):
    """Interface de storage de blobs.

    Local: AzuriteStorage. ProduÃ§Ã£o: a mesma ideia, trocando sÃ³ a credencial
    (connection string -> DefaultAzureCredential / Managed Identity). Nenhuma
    lÃ³gica de negÃ³cio referencia a impl concreta.
    """

    @abstractmethod
    def upload(self, path: str, data: bytes) -> str:
        """Sobe bytes para `path` no container; retorna o blob path gravado."""

    @abstractmethod
    def download(self, path: str) -> bytes:
        """Baixa o conteÃºdo do blob em `path`."""

    @abstractmethod
    def get_url(self, path: str) -> str:
        """URL de acesso ao blob (Azurite: http local; Azure: URL real)."""

    def ensure_container(self) -> None:
        """No-op por padrÃ£o. SÃ³ o Azurite precisa criar o container no boot;
        no Azure o container vem provisionado pela infra."""


class AzuriteStorage(BlobStorage):
    """Impl local - Azurite via azure-storage-blob (o mesmo SDK do Azure real)."""

    def __init__(
        self,
        connection_string: str | None = None,
        container: str | None = None,
    ) -> None:
        self._client = BlobServiceClient.from_connection_string(
            connection_string or settings.azure_storage_connection_string
        )
        self._container = container or settings.blob_container_name

    def ensure_container(self) -> None:
        """Cria o container se nÃ£o existir. Idempotente."""
        try:
            self._client.create_container(self._container)
        except ResourceExistsError:
            pass

    def upload(self, path: str, data: bytes) -> str:
        self._client.get_blob_client(self._container, path).upload_blob(data, overwrite=True)
        return path

    def download(self, path: str) -> bytes:
        return self._client.get_blob_client(self._container, path).download_blob().readall()

    def get_url(self, path: str) -> str:
        return self._client.get_blob_client(self._container, path).url

class AzureBlobStorage(BlobStorage):
    """Impl de produÃ§Ã£o: mesma conta de Blob, autenticada por Managed Identity
    (sem connection string). A MI precisa do papel "Storage Blob Data
    Contributor" na conta -- isso Ã© permissÃ£o na Azure, nÃ£o cÃ³digo."""

    def __init__(self, account_url: str | None = None, container: str | None = None) -> None:
        from azure.identity import DefaultAzureCredential

        self._client = BlobServiceClient(
            account_url or settings.azure_storage_account_url,
            credential=DefaultAzureCredential(),
        )
        self._container = container or settings.blob_container_name

    def upload(self, path: str, data: bytes) -> str:
        self._client.get_blob_client(self._container, path).upload_blob(data, overwrite=True)
        return path

    def download(self, path: str) -> bytes:
        return self._client.get_blob_client(self._container, path).download_blob().readall()

    def get_url(self, path: str) -> str:
        return self._client.get_blob_client(self._container, path).url
